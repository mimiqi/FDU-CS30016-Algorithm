# 正式题库与离线标准答案

标准答案只用于评分，严禁提供给被测模型。

## Q01 有限状态更新

```python
a = [2, 5, 1, 4, 3]
s = 0
for t in range(8):
    i = (2*t + s) % 5
    j = (i + 2) % 5
    a[i], a[j] = (a[j] + t) % 9, (a[i] + s) % 9
    s = (s + a[(i+1) % 5] - a[j]) % 13
print(a, s)
```

标准输出：

```text
[8, 3, 0, 2, 5] 10
```

## Q03 闭包与可变默认值

```python
def make():
    out = []
    box = [1]
    for i in range(4):
        def f(x=i, b=box):
            b[0] += x + i
            return b[0]
        out.append(f)
        if i % 2 == 0:
            box = [box[0] + i]
    return out
fs = make()
print([fs[k]() for k in [2, 0, 3, 1, 2, 0]])
print([f() for f in fs])
```

标准输出：

```text
[6, 4, 9, 10, 15, 7]
[10, 19, 24, 15]
```

## Q05 finally与返回引用

def f(n):
    try:
        if n % 3 == 0:
            return log[:]
        if n % 3 == 1:

        return len(log)
    finally:
        log.append(n)
        if n == 5:

## Q06 递归与共享缓存

```python
def f(n, memo={0: 1}):
    if n in memo:
        return memo[n]
    memo[n] = f(n-1) + (f(n-2) if n > 1 else 0)
    if n % 2 == 0:
        memo[n-1] += 1
    return memo[n]
print(f(6), f(3), f(5))
print(f(7), f(4), f(1))
```

标准输出：

```text
17 5 12
29 6 2
```

## Q10 生成器暂停与finally

```python
log = []
def g():
    try:
        for i in range(3):
            log.append(i)
            x = yield i + len(log)
            log.append(-1 if x is None else x)
    finally:
        log.append(9)
h = g()
r = [next(h), h.send(4), next(h)]
h.close()
print(r, log)
```

标准输出：

```text

```

## Q11 嵌套循环与位运算

```python
s = 3
r = []
for i in range(1, 5):
    for j in range(i, -1, -1):
        if (s + j) % 3 == 0:
            s ^= (i << 1) + j
            continue
        s = (s*2 + j - i) % 19
        if s % 5 == 0:
            break
    r.append(s)
print(r, s)
```

标准输出：

```text
[4, 12, 10, 0] 0
```

## Q19 双向状态变换

```python
a, b, c = 2, 3, 5
r = []
for k in range(1, 9):
    a, b = (a+b+k) % 11, (a*2-c) % 11
    if b < a:
        c = (c+b-k) % 13
    else:
        a, c = c % 11, (a+c) % 13
    r.append((a, b, c))
print(r)
```

标准输出：

```text
[(5, 10, 11), (0, 10, 4), (4, 7, 6), (4, 2, 4), (4, 4, 4), (4, 4, 7), (4, 1, 1), (1, 7, 3)]
```

## Q20 异常控制流

```python
r = []
for n in range(5):
    try:
        r.append(10 // (n-2))
        if n % 2:
            continue
        r.append(n)
    except ZeroDivisionError:
        r.append(7)
    else:
        r.append(8)
    finally:
        r.append(-n)
print(r)
```

标准输出：

```text
[-5, 0, 8, 0, -10, -1, 7, -2, 10, -3, 5, 4, 8, -4]
```
