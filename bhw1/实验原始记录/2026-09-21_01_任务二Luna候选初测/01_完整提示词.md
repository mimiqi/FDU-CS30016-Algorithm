算法 A：
procedure A(n):
    for i = 2 to n:
        j = i
        while j <= n:
            work()
            if j > floor(n / j):
                break
            j = j * j

算法 B：
procedure B(n):
    if n <= 2:
        work()
        return
    B(floor(n / 2))
    B(ceil(n / 2))
    for i = 1 to floor(n / floor(log2(n))):
        work()

算法 C：
procedure C(n):
    if n <= 2:
        work()
        return
    k = floor(log2(n))
    r = floor(n / k)
    C(r)
    C(n - r)
    for i = 1 to n:
        work()
